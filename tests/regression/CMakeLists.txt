cmake_minimum_required(VERSION 3.20)
project(localvocal_regression LANGUAGES CXX)
enable_testing()
add_executable(localvocal_regression regression.cpp)
target_compile_features(localvocal_regression PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(localvocal_regression PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(localvocal_regression PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME translation_regression COMMAND localvocal_regression)
