cmake_minimum_required(VERSION 3.20)
project(localvocal_regression LANGUAGES CXX)
enable_testing()
add_executable(localvocal_regression regression.cpp)
target_compile_features(localvocal_regression PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(localvocal_regression PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(localvocal_regression PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME translation_regression COMMAND localvocal_regression)
find_package(Threads REQUIRED)
add_executable(localvocal_captions captions.cpp)
target_compile_features(localvocal_captions PRIVATE cxx_std_17)
target_link_libraries(localvocal_captions PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(localvocal_captions PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(localvocal_captions PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME caption_pipeline_regression COMMAND localvocal_captions)
